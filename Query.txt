CREATE TABLE STUDENT_TBL (
    STUDENT_REG_NO   VARCHAR2(15) PRIMARY KEY,
    FULL_NAME        VARCHAR2(100) NOT NULL,
    BRANCH           VARCHAR2(20) NOT NULL,
    CURRENT_SEMESTER NUMBER(2) CHECK (CURRENT_SEMESTER BETWEEN 1 AND 8),
    CGPA             NUMBER(3,2) CHECK (CGPA BETWEEN 0 AND 10),
    EMAIL            VARCHAR2(100) UNIQUE,
    MOBILE           VARCHAR2(15),
    STATUS           VARCHAR2(15) CHECK (STATUS IN ('ACTIVE','INACTIVE'))
);

CREATE TABLE APPLICATION_TBL (
    APPLICATION_ID     NUMBER PRIMARY KEY,
    STUDENT_REG_NO     VARCHAR2(15),
    COMPANY_ID         VARCHAR2(10),
    APPLIED_DATE       DATE,
    APPLICATION_STATUS VARCHAR2(30),
    INTERVIEW_DATE     DATE,
    INTERVIEW_SLOT     VARCHAR2(20),
    INTERVIEW_PANEL    VARCHAR2(100),
    FINAL_OUTCOME      VARCHAR2(30),

    CONSTRAINT FK_STUDENT
    FOREIGN KEY (STUDENT_REG_NO)
    REFERENCES STUDENT_TBL(STUDENT_REG_NO),

    CONSTRAINT FK_COMPANY
    FOREIGN KEY (COMPANY_ID)
    REFERENCES COMPANY_TBL(COMPANY_ID),

    CONSTRAINT UK_STUDENT_COMPANY
    UNIQUE (STUDENT_REG_NO, COMPANY_ID)
);
